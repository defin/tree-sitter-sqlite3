==============================================

WITH RECURSIVE counter (n) AS (SELECT 1 UNION ALL SELECT n + 1 FROM counter WHERE n < 10) SELECT * FROM counter;

---

; Compound SELECT inside the CTE body — the second arm of UNION ALL
; flattens into the same select_clause via repeat(...) rather than
; nesting a sub-select_clause node. Each arm contributes its own
; result_column_list / from / where fields to the outer node.

    with: (with_clause
      cte: (cte_definition

        column: (identifier)

          columns: (result_column_list

              expression: (numeric_literal)))

            (expression_result_column
              expression: (binary_expression
                left: (column_reference
                  name: (identifier))
                right: (numeric_literal))))
          from: (from_clause
            source: (table_or_subquery
              (qualified_table_name
                name: (identifier))))

            (binary_expression

                name: (identifier))
